    Which agents this controller uses

    List each worker PC. Tick the agents this controller should own, then Point ticked agents here. Two controllers can run at once — each ticks a different set, so each drives its own agents with no overlap. Untick an agent to leave it for the other controller. How it works ▸

    1. Each agent listens on port 4101 for a "switch controller" command (installer opens this port in Windows Firewall).
    2. This controller's address is auto-detected below — ticked agents get pointed here. Override only if this PC has several network cards.
    3. Two controllers: on PC-A tick agents 1–3, on PC-B tick agents 4–5. Each clicks "Point ticked agents here" and runs its own tests in parallel. Keep the ticked sets separate — if both tick the same agent, whoever clicks last takes it.
    4. An agent already running a test is skipped (shown as "busy") — you can't yank it mid-run.

    Google Sheets

    Auto-append each finished run's results as a row in your Google Sheet — no more manual entry. Show one-time setup steps ▸

    1. Open your Google Sheet → Extensions → Apps Script.
    2. Delete anything there and paste this, then Save. It auto-detects your layout: a tab named Main Report + one tab per person → fills A,B,D,E,F on Main and G,H,I,J on each person's tab (leaving your formula columns alone); otherwise it just appends one flat row to the first sheet. 
      function doPost(e) {
        var ss = SpreadsheetApp.getActiveSpreadsheet();
        var d = JSON.parse(e.postData.contents);
        var main = ss.getSheetByName('Main Report');
        if (main) {                                    // report layout: Main Report + one tab per person
          var R = nextRow_(main, 2);                   // next free row (anchored on col B = Action)
          main.getRange(R, 1).setValue(d.date);        // A  Date
          main.getRange(R, 2).setValue(d.action);      // B  Action / Section
          main.getRange(R, 4).setValue(d.totalThread); // D  Total threads (all agents)
          main.getRange(R, 5).setValue(d.ramp);        // E  Ramp
          main.getRange(R, 6).setValue(d.loop);        // F  Loop
          (d.persons || []).forEach(function (p) {
            var sh = findPersonSheet_(ss, p.person);
            if (!sh) return;
            sh.getRange(R, 7).setValue(p.errorRatePct);  // G  Error Rate %
            sh.getRange(R, 8).setValue(p.throughput);    // H  Throughput
            sh.getRange(R, 9).setValue(p.time);          // I  Time
            sh.getRange(R, 10).setValue(p.thread);       // J  individual Thread
          });
        } else {                                       // simple layout: one flat row on the first sheet
          var s0 = ss.getSheets()[0];
          if (s0.getLastRow() === 0) s0.appendRow(['Date','Plan','Target','Agents','Duration(s)','Samples','Errors','Error %','Avg','Median','p90','p95','p99','Min','Max','Throughput']);
          s0.appendRow([d.date, d.plan, d.target, d.agents, d.durationSec, d.samples, d.errors, d.errorPct, d.avg, d.median, d.p90, d.p95, d.p99, d.min, d.max, d.throughput]);
        }
        return ContentService.createTextOutput('ok');
      }

      function doGet() { return ContentService.createTextOutput('LoadPilot Google Sheets sync is connected.'); }

      function nextRow_(sh, col) {                      // last used row in `col`, + 1 (skips formula cells elsewhere)
        var v = sh.getRange(1, col, sh.getMaxRows(), 1).getValues(), r = 2;
        for (var i = 0; i < v.length; i++) if (v[i][0] !== '' && v[i][0] !== null) r = i + 2;
        return r;
      }

      function findPersonSheet_(ss, person) {           // match a person to its tab ("9-mizan" ~ "mizan")
        var norm = function (s) { return String(s).replace(/^\s*\d+\s*[-_. ]\s*/, '').trim().toLowerCase(); };
        var target = norm(person), hit = null;
        if (!target) return null;
        ss.getSheets().forEach(function (s) {
          if (s.getName() === 'Main Report') return;
          var n = norm(s.getName());
          if (n === target) hit = s;
          else if (!hit && n && (n.indexOf(target) >= 0 || target.indexOf(n) >= 0)) hit = s;
        });
        return hit;
      }
